// Isolated dual-width frontend: four registered groups, at most eight words.
// Two disjoint PC[2] BTB banks preserve the original 64-entry capacity.
// No request acceptance or consumer pop feeds input credit combinationally.
module nc_fetch #(parameter ENABLE_BP=1)(input wire clk,rst_n,redirect,stop,input wire[31:0] target,
 input wire bp_flush,bp_update_valid,bp_update_branch,bp_update_conditional,bp_update_taken,
 input wire[31:0] bp_update_pc,bp_update_target,output wire[31:0] prediction,prediction1,
 output wire req_valid,input wire req_ready,output wire[31:0] req_addr,
 input wire rsp_valid,output wire rsp_ready,input wire[63:0] rsp_data,input wire rsp_error,
 output wire[9:0] pair_flags,
 output wire valid,input wire ready,input wire take_two,output wire second_valid,
 output wire[31:0] pc,inst,inst1,output wire fault);
 reg[31:0] next_pc,cmd_pc,flight_pc,cmd_prediction,flight_prediction,cmd_prediction1,flight_prediction1;
 reg cmd_two,flight_two,cmd_valid,cmd_stale,flight,flight_stale;
 wire[31:0] lookup_prediction,lookup_prediction1;
 wire[31:0] even_prediction,odd_prediction;
 wire[31:0] even_pc={next_pc[31:3],3'b000},odd_pc={next_pc[31:3],3'b100};
 generate if(ENABLE_BP) begin:predictor
  nc_btb_bank btb(.clk(clk),.rst_n(rst_n),.flush(bp_flush),.lookup_pc(even_pc),.prediction(even_prediction),
   .update_valid(bp_update_valid && !bp_update_pc[2]),.update_branch(bp_update_branch),.update_conditional(bp_update_conditional),
   .update_taken(bp_update_taken),.update_pc(bp_update_pc),.update_target(bp_update_target));
  nc_btb_bank btb1(.clk(clk),.rst_n(rst_n),.flush(bp_flush),.lookup_pc(odd_pc),.prediction(odd_prediction),
   .update_valid(bp_update_valid && bp_update_pc[2]),.update_branch(bp_update_branch),.update_conditional(bp_update_conditional),
   .update_taken(bp_update_taken),.update_pc(bp_update_pc),.update_target(bp_update_target));
 assign lookup_prediction=next_pc[2] ? odd_prediction : even_prediction;
  assign lookup_prediction1=odd_prediction;
 end else begin:no_predictor
  assign lookup_prediction=0;assign lookup_prediction1=0;
 end endgenerate
 reg[63:0] instructions[0:3] /* synthesis syn_ramstyle="lut_ram" */;
 reg[33:0] metadata[0:3] /* synthesis syn_ramstyle="registers" */;
 reg[63:0] predictions[0:3] /* synthesis syn_ramstyle="registers" */;
 // {primary eligible, plain ALU legal, writes rd, use2, use1}.
 // These bits terminate at the same registered FIFO/cache boundary as
 // instructions, never through an EX/LSU-ready dependent predecode path.
 function[7:0] classify_parts;
  input[31:0] insn;
  reg[6:0] op,f7;reg[2:0] f3;
  reg upper,imm_op,reg_op,zero_f7,alt_f7;
  reg base_alu,zero_alu,alt_alu,memory_op,multiply_op;
  begin
   op=insn[6:0];f3=insn[14:12];f7=insn[31:25];
   upper=(op & 7'h5f)==7'h17;
   imm_op=op==7'h13;reg_op=op==7'h33;
   zero_f7=f7==0;alt_f7=f7==7'h20;
   // Non-shift immediates do not interpret upper immediate bits as f7.
   base_alu=upper || (imm_op && (f3[1] || !f3[0]));
   zero_alu=(imm_op || reg_op) && zero_f7;
   alt_alu=alt_f7 && ((imm_op && f3==5) || (reg_op && (f3==0 || f3==5)));
   memory_op=(op==7'h03 && (!f3[1] || f3==2)) ||
             (op==7'h23 && !f3[2] && f3[1:0]!=3);
   multiply_op=reg_op && f7==1 && !f3[2];
   classify_parts={multiply_op,memory_op,alt_alu,zero_alu,base_alu,
                   op!=7'h23,reg_op,imm_op || reg_op};
  end
 endfunction
 function[4:0] pack_flags;
  input[7:0] parts;
  begin pack_flags={|parts[7:3],|parts[5:3],parts[2:0]};end
 endfunction
 // Simulation observability only; synthesized incoming data uses parts.
 function[4:0] classify;
  input[31:0] insn;
  begin classify=pack_flags(classify_parts(insn));end
 endfunction
 reg[15:0] flags[0:3] /* synthesis syn_ramstyle="registers" */;
 reg[15:0] head_parts;assign pair_flags={pack_flags(head_parts[15:8]),pack_flags(head_parts[7:0])};
 wire[15:0] incoming_flags=flight_pc[2] ? {8'b0,classify_parts(rsp_data[63:32])} :
                                               {classify_parts(rsp_data[63:32]),classify_parts(rsp_data[31:0])};
 reg[63:0] head_inst,head_predictions;reg[33:0] head_meta;
 // Registered following-group cache. Instruction RAM never feeds ID
 // directly. The FIFO retains four groups and its original credit rule.
 reg[63:0] peek_inst,peek_predictions;reg[15:0] peek_parts;reg[33:0] peek_meta;
 reg[1:0] rd,wr;reg[2:0] count;
 wire group_two=head_meta[33];
 // A high-word fall-through command is followed by the next aligned group.
 // A taken prediction disables this window; redirects discard the queue.
 // Require two words in the following group so consuming its low word
 // leaves one group at the new head, without a two-entry pop operation.
 reg head_cross;wire cross_window=head_cross;
 wire[1:0] after_following=rd+2'd2;
 wire response=flight && rsp_valid;
 assign req_valid=cmd_valid && (!flight || response);assign req_addr=cmd_pc;assign rsp_ready=flight;
 wire launch=req_valid && req_ready;
 wire[3:0] reserved={1'b0,count}+{3'b0,flight}+{3'b0,cmd_valid};
 wire prepare=(!cmd_valid || launch) && reserved<4 && !stop && !redirect;
 wire push=response && !flight_stale && !redirect;
 wire consume=valid && ready;
 wire pop=consume && (!group_two || take_two);
 assign valid=count!=0;assign second_valid=group_two || cross_window;assign {fault,pc}=head_meta[32:0];
 assign inst=head_inst[31:0];assign inst1=head_inst[63:32];
 assign prediction=head_predictions[31:0];assign prediction1=head_predictions[63:32];

 wire[63:0] incoming=flight_pc[2] ? {32'b0,rsp_data[63:32]} : rsp_data;
 wire[33:0] incoming_meta={flight_two,rsp_error,flight_pc};
 wire[63:0] incoming_predictions={flight_prediction1,flight_prediction};
 wire prepare_two=!next_pc[2] && !lookup_prediction[30] && !next_pc[1] && !next_pc[0];
 always @(posedge clk) begin
  if(pop && count>2) begin
   peek_inst<=instructions[after_following];peek_parts<=flags[after_following];
   peek_meta<=metadata[after_following];peek_predictions<=predictions[after_following];
  end else if(push && ((!pop && count==1) || (pop && count==2))) begin
   peek_inst<=incoming;peek_parts<=incoming_flags;peek_meta<=incoming_meta;peek_predictions<=incoming_predictions;
  end
 end
 // This function forms a view BEFORE the head registers. Instruction
 // class, instruction bits, prediction and cross ownership are all sampled
 // on the same edge. ID sees no head/peek data mux or late fault gate.
 function[178:0] frame;
  input[15:0] gp,np;input[63:0] gi,ni;input[33:0] gm,nm;
  input[63:0] gr,nr;input nv;
  reg cross;
  begin
   cross=!gm[33] && gm[2] && !gm[32] && !gr[30] && nv && nm[33] && !nm[32];
   frame={cross,(cross ? {np[7:0],gp[7:0]} : gp),
                (cross ? {ni[31:0],gi[31:0]} : gi),gm,
                (cross ? {nr[31:0],gr[31:0]} : gr)};
  end
 endfunction
 wire near_valid=count>1 || push;
 wire[15:0] near_parts=count>1 ? peek_parts : incoming_flags;
 wire[63:0] near_inst=count>1 ? peek_inst : incoming;
 wire[33:0] near_meta=count>1 ? peek_meta : incoming_meta;
 wire[63:0] near_predictions=count>1 ? peek_predictions : incoming_predictions;
 wire far_valid=count>2 || (count==2 && push);
 wire[15:0] far_parts=count>2 ? flags[after_following] : incoming_flags;
 wire[63:0] far_inst=count>2 ? instructions[after_following] : incoming;
 wire[33:0] far_meta=count>2 ? metadata[after_following] : incoming_meta;
 wire[63:0] far_predictions=count>2 ? predictions[after_following] : incoming_predictions;
 always @(posedge clk) begin
  if(push) begin flags[wr]<=incoming_flags;instructions[wr]<=incoming;metadata[wr]<=incoming_meta;predictions[wr]<=incoming_predictions;end
  if(consume) begin
   if(group_two && !take_two) begin
    {head_cross,head_parts,head_inst,head_meta,head_predictions}<=frame(
     {8'b0,head_parts[15:8]},near_parts,{32'b0,head_inst[63:32]},near_inst,
     {1'b0,head_meta[32],pc+32'd4},near_meta,{32'b0,head_predictions[63:32]},near_predictions,near_valid);
   end else if(cross_window && take_two) begin
    {head_cross,head_parts,head_inst,head_meta,head_predictions}<=frame(
     {8'b0,peek_parts[15:8]},far_parts,{32'b0,peek_inst[63:32]},far_inst,
     {1'b0,peek_meta[32],peek_meta[31:0]+32'd4},far_meta,{32'b0,peek_predictions[63:32]},far_predictions,far_valid);
   end else if(count>1) begin
    {head_cross,head_parts,head_inst,head_meta,head_predictions}<=frame(
     peek_parts,far_parts,peek_inst,far_inst,peek_meta,far_meta,peek_predictions,far_predictions,far_valid);
   end else begin
    {head_cross,head_parts,head_inst,head_meta,head_predictions}<=frame(
     push ? incoming_flags : 16'b0,16'b0,push ? incoming : 64'b0,64'b0,
     push ? incoming_meta : 34'b0,34'b0,push ? incoming_predictions : 64'b0,64'b0,1'b0);
   end
  end else if(push && count==0) begin
   {head_cross,head_parts,head_inst,head_meta,head_predictions}<=frame(
    incoming_flags,16'b0,incoming,64'b0,incoming_meta,34'b0,incoming_predictions,64'b0,1'b0);
  end else if(push && count==1) begin
   {head_cross,head_parts,head_inst,head_meta,head_predictions}<=frame(
    head_parts,incoming_flags,head_inst,incoming,head_meta,incoming_meta,head_predictions,incoming_predictions,1'b1);
  end
 end
 always @(posedge clk or negedge rst_n) begin
  if(!rst_n) begin
   next_pc<=0;cmd_pc<=0;flight_pc<=0;cmd_valid<=0;cmd_stale<=0;flight<=0;flight_stale<=0;
   cmd_prediction<=0;flight_prediction<=0;cmd_prediction1<=0;flight_prediction1<=0;
   cmd_two<=0;flight_two<=0;rd<=0;wr<=0;count<=0;
  end else begin
   if(launch) begin
    cmd_valid<=0;flight<=1;flight_pc<=cmd_pc;flight_prediction<=cmd_prediction;
    flight_prediction1<=cmd_prediction1;flight_two<=cmd_two;flight_stale<=cmd_stale || redirect;
   end else if(response) flight<=0;
   if(prepare) begin
    cmd_valid<=1;cmd_pc<=next_pc;cmd_prediction<=lookup_prediction;cmd_prediction1<=lookup_prediction1;
    cmd_two<=prepare_two;cmd_stale<=0;
    next_pc<=lookup_prediction[30] ? {lookup_prediction[29:0],2'b0} :
      prepare_two && lookup_prediction1[30] ? {lookup_prediction1[29:0],2'b0} : next_pc+(prepare_two ? 32'd8 : 32'd4);
   end
   if(push) wr<=wr+1'b1;if(pop) rd<=rd+1'b1;
   case({push,pop}) 2'b10:count<=count+1'b1;2'b01:count<=count-1'b1;default:;endcase
   if(redirect) begin
    rd<=0;wr<=0;count<=0;next_pc<=target;
    if(cmd_valid && !launch) cmd_stale<=1;
    else begin
     cmd_valid<=1;cmd_pc<=target;cmd_prediction<=0;cmd_prediction1<=0;cmd_stale<=0;
     cmd_two<=!target[2] && !target[1] && !target[0];
     next_pc<=target+((!target[2] && !target[1] && !target[0]) ? 32'd8 : 32'd4);
    end
    if(flight && !launch) flight_stale<=1;
   end
  end
 end
endmodule
